// Registration is explicit and per render; extensions never receive parser/visitor objects.
public sealed class TemplateExtensions
{
    private readonly Dictionary<string, Func<ExtensionCall, TemplateValue>> methods = new(StringComparer.Ordinal);
    private readonly Dictionary<string, Func<ValueLookup, TemplateValue>> values = new(StringComparer.Ordinal);
    public TemplateExtensions AddMethod(string name, Func<ExtensionCall, TemplateValue> method)
    {
        ArgumentException.ThrowIfNullOrEmpty(name);
        ArgumentNullException.ThrowIfNull(method);
        if (name.StartsWith('@')) throw new ArgumentException("Annotation overrides are not supported; register value functions with AddValueFunction.", nameof(name));
        methods.Add(name, method);
        return this;
    }
    public TemplateExtensions AddValueFunction(string name, Func<ValueLookup, TemplateValue> function)
    {
        ArgumentException.ThrowIfNullOrEmpty(name);
        ArgumentNullException.ThrowIfNull(function);
        values.Add(name, function);
        return this;
    }
    internal bool TryMethod(string name, out Func<ExtensionCall, TemplateValue> method) => methods.TryGetValue(name, out method!);
    internal bool TryValueFunction(string name, out Func<ValueLookup, TemplateValue> function) => values.TryGetValue(name, out function!);
    internal TemplateExtensions Snapshot()
    {
        var copy = new TemplateExtensions();
        foreach (var entry in methods) copy.methods.Add(entry.Key, entry.Value);
        foreach (var entry in values) copy.values.Add(entry.Key, entry.Value);
        return copy;
    }
}

// Implement Configure in a handwritten partial class outside Generated/.
public static partial class TemplateExtensionRegistration
{
    static partial void Configure(TemplateExtensions extensions);
    public static TemplateExtensions Create()
    {
        var extensions = new TemplateExtensions();
        Configure(extensions);
        return extensions;
    }
}

[System.Diagnostics.DebuggerDisplay("{Count} deferred arguments")]
public sealed class ExtensionArguments
{
    private readonly EvaluationContext context;
    private readonly Func<object?>[] arguments;
    internal ExtensionArguments(EvaluationContext context, Func<object?>[] arguments)
    { this.context = context; this.arguments = arguments; }
    public int Count => arguments.Length;
    // Deliberately not cached: repeated evaluation is an explicit extension decision.
    public TemplateValue Evaluate(int index) => new(context, arguments[index]());
}

public sealed class ExtensionContext
{
    private readonly EvaluationContext context;
    internal ExtensionContext(EvaluationContext context) => this.context = context;
    public TemplateValue Lookup(string path) => new(context, context.Lookup(path));
    public override string ToString() => context.ToString();
}

public sealed class ExtensionCall
{
    private readonly EvaluationContext context;
    internal ExtensionCall(EvaluationContext context, object? receiver, Func<object?>[] arguments)
    {
        this.context = context;
        Receiver = new(context, receiver);
        Arguments = new(context, arguments);
        Context = new(context);
    }
    public TemplateValue Receiver { get; }
    public ExtensionArguments Arguments { get; }
    public ExtensionContext Context { get; }
    public void ReportError(string message,
        [System.Runtime.CompilerServices.CallerFilePath] string file = "",
        [System.Runtime.CompilerServices.CallerLineNumber] int line = 0) => context.ReportError(message, file + ":" + line);
    public void RequireArgumentCount(int count,
        [System.Runtime.CompilerServices.CallerFilePath] string file = "",
        [System.Runtime.CompilerServices.CallerLineNumber] int line = 0)
    {
        if (Arguments.Count != count) throw new ArgumentException(file + ":" + line + ": Expected " + count + " arguments.");
    }
}

public sealed class ValueLookup
{
    private readonly EvaluationContext context;
    internal ValueLookup(EvaluationContext context, string name) { this.context = context; Name = name; }
    public string Name { get; }
    public TemplateValue DefaultValue() => new(context, context.LookupDefault(Name));
    public override string ToString() => "Value lookup: " + Name;
}
