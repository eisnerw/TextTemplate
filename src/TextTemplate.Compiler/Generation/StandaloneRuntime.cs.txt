// Context evaluation and structured composition adapted from TextTemplateInterpreter.ts
// (eisnerw/texttemplate-editor, Apache-2.0, ccedefd87b012ff1173ee2f43323b6eb7c8bd268).
// Emitted as source: no compiler, parser, or TextTemplate assembly is needed at runtime.
[System.Diagnostics.DebuggerDisplay("{ToString(),nq}")]
internal sealed class EvaluationContext
{
    [System.Diagnostics.DebuggerDisplay("{ToString(),nq}")]
    internal sealed class Data(Dictionary<string, object?> values, Data? parent)
    {
        internal Dictionary<string, object?> Values { get; } = values;
        internal Data? Parent { get; } = parent;
        // Null denotes a dictionary; a list keeps its children and their parent links.
        internal List<Data>? Items { get; set; }
        public override string ToString() => DebugSummary(this);
    }
    // Debug summaries inspect stored values only: no lookup, rendering, callbacks,
    // or parent traversal. Bound depth and collection size for large/cyclic data.
    private static string DebugSummary(object? value)
    {
        object? Preview(object? item, int depth)
        {
            if (item is null) return null;
            if (depth >= 4) return "…";
            if (item is Data data) return data.Items is null ? ObjectPreview(data.Values, depth) : ArrayPreview(data.Items, depth);
            if (item is Missing missing) return missing.ToString();
            if (item is RenderedList rendered) return ArrayPreview(rendered.Items, depth);
            if (item is Arguments arguments) return ArrayPreview(arguments.Items, depth);
            if (item is string text) return text.Length <= 160 ? text : text[..160] + "…";
            if (item is JsonElement json)
                return json.ValueKind switch
                {
                    JsonValueKind.Object => ObjectPreview(Properties(json), depth),
                    JsonValueKind.Array => ArrayPreview(Elements(json), depth),
                    JsonValueKind.String => Preview(json.GetString(), depth),
                    JsonValueKind.Undefined => "<undefined>",
                    _ => json
                };
            if (item is TemplateContent content) return ArrayPreview(content.Fragments, depth);
            if (item is object?[] array) return ArrayPreview(array, depth);
            return item is bool or double or int or long ? item : "<" + item.GetType().Name + ">";
        }
        Dictionary<string, object?> ObjectPreview(IEnumerable<KeyValuePair<string, object?>> entries, int depth)
        {
            Dictionary<string, object?> result = new Dictionary<string, object?>(StringComparer.Ordinal);
            foreach (KeyValuePair<string, object?> entry in entries)
            {
                if (result.Count == 8) { result["…"] = "more properties"; break; }
                result[entry.Key] = Preview(entry.Value, depth + 1);
            }
            return result;
        }
        List<object?> ArrayPreview(IEnumerable<object?> entries, int depth)
        {
            List<object?> result = new List<object?>();
            foreach (object? entry in entries)
            {
                if (result.Count == 8) { result.Add("…"); break; }
                result.Add(Preview(entry, depth + 1));
            }
            return result;
        }
        static IEnumerable<KeyValuePair<string, object?>> Properties(JsonElement json)
        {
            foreach (JsonProperty property in json.EnumerateObject()) yield return new(property.Name, property.Value);
        }
        static IEnumerable<object?> Elements(JsonElement json)
        {
            foreach (JsonElement element in json.EnumerateArray()) yield return element;
        }
        return JsonSerializer.Serialize(Preview(value, 0), new JsonSerializerOptions
        {
            Encoder = System.Text.Encodings.Web.JavaScriptEncoder.UnsafeRelaxedJsonEscaping,
            NumberHandling = System.Text.Json.Serialization.JsonNumberHandling.AllowNamedFloatingPointLiterals
        });
    }
    public override string ToString()
        => $"Context {DebugSummary(Current)} (locals: {Locals.Count}, shared: {Shared.Count})";
    // TODO: implement each feature at its emitted call site. Keep operands deferred:
    // annotations and short-circuiting must not evaluate receiver/arguments prematurely.
    internal static object? NotImplemented(string feature, params Func<object?>[] operands)
        => throw new NotImplementedException("Execution is not implemented yet: " + feature);
    internal static object? At(string location, Func<object?> action)
    {
        try { return action(); }
        catch (NotImplementedException error) when (!error.Data.Contains("TemplateLocation"))
        {
            NotImplementedException located = new NotImplementedException(location + ": " + error.Message, error);
            located.Data["TemplateLocation"] = location;
            throw located;
        }
    }
    [System.Diagnostics.DebuggerDisplay("{ToString(),nq}")]
    internal sealed record Missing(string Key, string? Replacement = null)
    {
        public override string ToString() => Replacement is null
            ? $"Missing({JsonSerializer.Serialize(Key)})"
            : $"Missing({JsonSerializer.Serialize(Key)}, replacement: {JsonSerializer.Serialize(Replacement)})";
    }
    private readonly string? dataRoot;
    private string? missingText;
    private TemplateRegex? falsy;
    // TODO: Apply MultilineStyle during composition; currently only stored and inherited.
    internal IReadOnlyList<string>? MultilineStyle { get; private set; }
    private Missing MissingValue(string key) => new(key, missingText?.Replace("{key}", key));
    internal readonly Data? Current;
    internal readonly Dictionary<string, Func<EvaluationContext, object?>> Shared;
    internal Dictionary<string, Func<EvaluationContext, object?>> Locals = new(StringComparer.Ordinal);
    private sealed class CallDepth { internal int Value; }
    private readonly CallDepth recursion = new();

    private EvaluationContext(EvaluationContext parent, Data? current)
    {
        Current = current;
        Shared = parent.Shared;
        dataRoot = parent.dataRoot;
        Locals = new(parent.Locals, StringComparer.Ordinal);
        recursion = parent.recursion;
        missingText = parent.missingText;
        falsy = parent.falsy;
        MultilineStyle = parent.MultilineStyle;
    }

    [System.Runtime.CompilerServices.InterpolatedStringHandler]
    [System.Diagnostics.DebuggerDisplay("{ToString(),nq}")]
    internal struct StructuredText
    {
        private readonly List<object?> fragments;
        public StructuredText(int literalLength, int formattedCount) => fragments = new();
        public void AppendLiteral(string text) => fragments.Add(text);
        public void AppendFormatted(object? value) => fragments.Add(value);
        internal object?[] ToFragments() => fragments.ToArray();
        public override string ToString() => DebugSummary(fragments?.ToArray());
    }
    internal static TemplateContent Text(StructuredText text) => new(text.ToFragments());

    internal EvaluationContext(JsonElement? data, Dictionary<string, Func<EvaluationContext, object?>> shared, string? dataRoot = null)
    {
        Shared = shared;
        this.dataRoot = dataRoot;
        if (data.HasValue && data.Value.ValueKind != JsonValueKind.Null) Current = ReadData(data.Value, null);
    }
    private static Data ReadData(JsonElement value, Data? parent)
    {
        // The legacy TemplateData constructor collapses single-element arrays.
        if (value.ValueKind == JsonValueKind.Array && value.GetArrayLength() == 1) return ReadData(value[0], parent);
        Data data = new Data(new(StringComparer.Ordinal), parent);
        if (value.ValueKind == JsonValueKind.Array)
        {
            data.Items = new();
            foreach (JsonElement item in value.EnumerateArray()) data.Items.Add(ReadData(item, data));
        }
        else if (value.ValueKind == JsonValueKind.Object)
        {
            foreach (JsonProperty property in value.EnumerateObject())
                if (property.Value.ValueKind != JsonValueKind.Array || property.Value.GetArrayLength() != 0)
                    data.Values[property.Name] = property.Value.Clone();
        }
        else if (value.ValueKind == JsonValueKind.String) data.Values["_"] = value.GetString();
        else if (value.ValueKind == JsonValueKind.Null) throw new InvalidOperationException("Null list data is not a legacy data context.");
        return data;
    }
    internal object? Lookup(string path)
    {
        object? Find(Data data, string key)
        {
            int dot = key.IndexOf('.');
            string head = dot < 0 ? key : key[..dot];
            object? value;
            if (data.Values.TryGetValue(head, out object? found)) value = Materialize(found, data);
            else if (head == "*") value = data;
            else if (head == "^") value = data.Parent ?? data;
            else if (data.Items is not null)
            {
                List<object?> values = new List<object?>();
                foreach (Data item in data.Items)
                {
                    object? candidate = Find(item, key);
                    if (Condition(candidate)) values.Add(candidate);
                }
                return values.Count switch { 0 => MissingValue(path), 1 => values[0], _ => ScalarData(values) };
            }
            else return MissingValue(path);
            return dot < 0 ? value : value is Data nested ? Find(nested, key[(dot + 1)..]) : MissingValue(path);
        }
        object? value = Current is null ? MissingValue(path) : Find(Current, path);
        if (value is string text) return text.Length == 0 ? MissingValue(path) : text.Replace("\r", "");
        return value;
    }
    private static Data ScalarData(List<object?> values)
    {
        Data data = new Data(new(StringComparer.Ordinal), null) { Items = new() };
        foreach (object? value in values)
        {
            if (value is Data child) data.Items.Add(child);
            else
            {
                Data item = new Data(new(StringComparer.Ordinal), data);
                if (value is string) item.Values["_"] = value;
                data.Items.Add(item);
            }
        }
        return data;
    }
    private static object? Materialize(object? value, Data parent)
    {
        if (value is not JsonElement json) return value;
        return json.ValueKind switch
        {
            JsonValueKind.String => json.GetString(),
            JsonValueKind.Number => json.GetDouble(),
            JsonValueKind.True => true, JsonValueKind.False => false, JsonValueKind.Null => null,
            JsonValueKind.Object or JsonValueKind.Array => ReadData(json, parent),
            _ => null
        };
    }
    // Template output keeps its fragments; method receivers keep composed item values.
    // Neither is an object-data context, and neither is flattened before composition.
    [System.Diagnostics.DebuggerDisplay("{ToString(),nq}")]
    internal sealed record RenderedList(List<object?> Items)
    {
        public override string ToString() => DebugSummary(this);
    }
    [System.Diagnostics.DebuggerDisplay("{ToString(),nq}")]
    internal sealed record Arguments(List<object?> Items)
    {
        public override string ToString() => DebugSummary(this);
    }
    internal object? Each(Func<EvaluationContext, object?> body)
    {
        if (Current?.Items is not { } items) return body(this);
        List<object?> results = new List<object?>();
        foreach (Data item in items)
        {
            object? result = body(new EvaluationContext(this, item));
            if (Compose(result) is not null) results.Add(result);
        }
        return results.Count switch { 0 => MissingValue("list"), 1 => results[0], _ => new RenderedList(results) };
    }
    internal object? Receiver(Func<EvaluationContext, object?> body)
    {
        if (Current?.Items is not { } items) return body(this);
        List<object?> results = new List<object?>();
        foreach (Data item in items) results.Add(Intermediate(body(new EvaluationContext(this, item))));
        return new Arguments(results);
    }
    internal object? Context(object? value, Func<EvaluationContext, object?> body, Func<object?>? fallback = null)
    {
        value = Intermediate(value);
        if (value is string text)
        {
            if (text.StartsWith('/') || text.StartsWith("http", StringComparison.OrdinalIgnoreCase))
            {
                if (dataRoot is null) throw new InvalidOperationException("Specify --data-root DIRECTORY to resolve " + text + ".");
                if (!text.StartsWith("/data/", StringComparison.Ordinal) || text[6..].IndexOfAny(new[] { '/', '?', '#' }) >= 0)
                    throw new NotImplementedException("Unsupported resource path: " + text);
                string name = Uri.UnescapeDataString(text[6..]);
                if (name.Length == 0 || name is "." or ".." || name.IndexOfAny(new[] { '/', '\\' }) >= 0)
                    throw new InvalidOperationException("Invalid data resource name: " + name);
                string path = Path.Combine(dataRoot, name + ".json");
                if (!File.Exists(path)) throw new InvalidOperationException("Data " + name + " not found");
                using JsonDocument resource = JsonDocument.Parse(File.ReadAllText(path));
                JsonElement loaded = resource.RootElement;
                if (loaded.ValueKind is JsonValueKind.Null or JsonValueKind.False ||
                    (loaded.ValueKind == JsonValueKind.String && loaded.GetString() == "") ||
                    (loaded.ValueKind == JsonValueKind.Number && loaded.GetDouble() == 0))
                    throw new InvalidOperationException("Data " + name + " not found");
                if (loaded.ValueKind == JsonValueKind.Object && loaded.TryGetProperty("error", out JsonElement error))
                    throw new InvalidOperationException(error.ToString());
                if (loaded.ValueKind == JsonValueKind.Object && loaded.TryGetProperty("Result", out JsonElement result) && result.ValueKind == JsonValueKind.String && result.GetString() != "")
                    loaded = JsonSerializer.SerializeToElement(result.GetString()!.Replace("\r", ""));
                if (loaded.ValueKind == JsonValueKind.String && loaded.GetString() is string json && (json.StartsWith('{') || json.StartsWith('[')))
                {
                    using JsonDocument nested = JsonDocument.Parse(json);
                    value = ReadData(nested.RootElement, Current);
                }
                else value = ReadData(loaded, Current);
                return body(new EvaluationContext(this, (Data)value));
            }
            if (!text.StartsWith('{') && !text.StartsWith('[')) return text;
            using JsonDocument parsed = JsonDocument.Parse(text);
            value = ReadData(parsed.RootElement, Current);
        }
        if (value is Missing) return fallback is null ? value : fallback();
        if (value is not Data) throw new NotImplementedException("Expected an object data context.");
        return body(new EvaluationContext(this, (Data)value));
    }
    // Adapted from invokeMethods, visitNamedSubtemplate and callMethod in the preserved interpreter.
    // Annotations run before their receiver, with exported include definitions scoped to that call.
    internal EvaluationContext Define(Dictionary<string, Func<EvaluationContext, object?>> definitions)
    {
        EvaluationContext child = new EvaluationContext(this, Current);
        foreach (KeyValuePair<string, Func<EvaluationContext, object?>> entry in definitions) child.Locals[entry.Key] = entry.Value;
        return child;
    }
    internal EvaluationContext WithFalsy(TemplateRegex? pattern)
    {
        EvaluationContext child = new EvaluationContext(this, Current);
        child.falsy = pattern;
        return child;
    }
    internal bool TestCondition(object? value)
        => Condition(value) && !(falsy?.Test(Builtins.JsString(value)) ?? false);
    internal EvaluationContext WithMultilineStyle(params string[] styles)
    {
        bool tabbed = Array.IndexOf(styles, "Tabbed") >= 0;
        foreach (string style in styles)
            if (style is not ("Indented" or "IndentAllButFirst" or "Padded" or "Tabbed" or "Trimmed" or "Offset") ||
                (tabbed && style is ("Indented" or "IndentAllButFirst")))
                throw new InvalidOperationException("ERROR: invalid argument for multiline style");
        EvaluationContext child = new EvaluationContext(this, Current);
        child.MultilineStyle = styles.Length == 0 ? null : Array.AsReadOnly((string[])styles.Clone());
        return child;
    }
    internal EvaluationContext WithMissingValue(string replacement)
    {
        EvaluationContext child = new EvaluationContext(this, Current);
        child.missingText = replacement;
        return child;
    }
    internal object? Annotate(Func<EvaluationContext, EvaluationContext>[] annotations, Func<EvaluationContext, object?> body)
    {
        EvaluationContext child = new EvaluationContext(this, Current);
        foreach (Func<EvaluationContext, EvaluationContext> annotation in annotations) child = annotation(child);
        return body(child);
    }
    internal EvaluationContext Include(string name, Func<EvaluationContext, EvaluationContext> load)
    {
        if (Locals.ContainsKey(name)) throw new NotImplementedException("Locally bound include units are not implemented yet: " + name);
        if (++recursion.Value > 20) { recursion.Value--; throw new InvalidOperationException("Include recursion exceeds 20 levels."); }
        try { return load(new EvaluationContext(this, Current)); }
        finally { recursion.Value--; }
    }
    internal object? CallNamed(string name, object? value, params object?[] arguments)
    {
        value = Intermediate(value);
        Data data = new Data(new(StringComparer.Ordinal), Current);
        data.Values["$0"] = value;
        for (int i = 0; i < arguments.Length; i++)
        {
            object? argument = Intermediate(arguments[i]);
            if (argument is not Missing and not TemplateRegex) data.Values["$" + (i + 1)] = argument;
        }
        return new EvaluationContext(this, data).Named(name);
    }
    internal static object? JoinValues(object? value, params object?[] arguments)
    {
        if (arguments.Length > 2) throw new NotImplementedException("Join with more than two arguments is not implemented.");
        value = Intermediate(value);
        List<object?>? items = value is Arguments argumentsList ? argumentsList.Items : null;
        if (value is Data data)
        {
            if (data.Items is null) throw new NotImplementedException("Join requires a scalar list.");
            items = new();
            foreach (Data item in data.Items)
            {
                if (!item.Values.TryGetValue("_", out object? scalar) || !Condition(scalar))
                    throw new NotImplementedException("Join requires a scalar list.");
                items.Add(scalar);
            }
        }
        if (items is null) return value;
        List<string> kept = new List<string>();
        foreach (object? item in items) if (Condition(item)) kept.Add(Compose(item) ?? "");
        string separator = arguments.Length > 0 ? Compose(arguments[0]) ?? "" : ", ";
        string lastSeparator = arguments.Length > 1 ? Compose(arguments[1]) ?? "" : separator;
        if (kept.Count < 2) return string.Join("", kept);
        return string.Join(separator, kept.GetRange(0, kept.Count - 1)) + lastSeparator + kept[^1];
    }
    internal TResult Scope<TResult>(Dictionary<string, Func<EvaluationContext, object?>> definitions, Func<EvaluationContext, TResult> body)
    {
        EvaluationContext child = new EvaluationContext(this, Current);
        foreach (KeyValuePair<string, Func<EvaluationContext, object?>> entry in definitions) child.Locals[entry.Key] = entry.Value;
        return body(child);
    }
    internal object? Named(string name)
    {
        if (!Locals.TryGetValue(name, out Func<EvaluationContext, object?>? method) && !Shared.TryGetValue(name, out method))
            throw new InvalidOperationException("Subtemplate #" + name + " not found.");
        if (++recursion.Value > 20) { recursion.Value--; throw new InvalidOperationException("Subtemplate recursion exceeds 20 levels."); }
        try { return method(this); }
        finally { recursion.Value--; }
    }
    internal static object? Intermediate(object? value)
    {
        if (value is TemplateContent content)
            return content.Fragments.Length == 1 ? Intermediate(content.Fragments[0]) : Compose(content);
        if (value is RenderedList) return Compose(value);
        return value is object?[] sequence ? sequence.Length == 1 ? Intermediate(sequence[0]) : Compose(value) : value;
    }
    internal static bool Condition(object? value)
    {
        value = Intermediate(value);
        return value switch { null or Missing => false, string text => text.Length > 0,
            bool flag => flag, double number => number != 0 && !double.IsNaN(number), _ => true };
    }
    internal static bool Compare(object? left, string operation, object? right)
    {
        left = Intermediate(left); right = Intermediate(right);
        bool lm = left is null or Missing, rm = right is null or Missing;
        // Preserve the legacy missing-value comparison behavior, including its unusual cases.
        if (lm) return rm || operation == "!=";
        if (rm) return operation == "!=";
        string a = Scalar(left!), b = Scalar(right!);
        int? comparison = Builtins.CompareStrings(a, b);
        if (comparison is null) return operation == "!=";
        int order = comparison.Value;
        return operation switch { "=" or "==" => order == 0, "!=" => order != 0, ">" => order > 0,
            "<" => order < 0, ">=" => order >= 0, "<=" => order <= 0, _ => throw new InvalidOperationException(operation) };
    }
    internal static double ParseInteger(string text)
    {
        Match match = Regex.Match(text.TrimStart(Builtins.Whitespace.ToCharArray()), @"^[+-]?[0-9]+");
        return match.Success ? double.Parse(match.Value, CultureInfo.InvariantCulture) : double.NaN;
    }
    internal static string Scalar(object value) => value switch
    {
        string text => text, bool flag => flag ? "true" : "false",
        double number => NumberText(number),
        _ => throw new NotImplementedException("Non-scalar composition requires list/data formatting, which is not implemented yet.")
    };
    private static string NumberText(double number)
    {
        string text = number.ToString("R", CultureInfo.InvariantCulture);
        if (number == 0) return "0";
        int e = text.IndexOf('E');
        if (e < 0) return text;
        int exponent = int.Parse(text[(e + 1)..], CultureInfo.InvariantCulture);
        if (exponent >= -6 && exponent < 21) return number.ToString("0.####################", CultureInfo.InvariantCulture);
        return text[..e] + "e" + (exponent >= 0 ? "+" : "-") + Math.Abs(exponent).ToString(CultureInfo.InvariantCulture);
    }
    // Retain the fragment hierarchy until rendering; a missing value removes its entire line.
    internal static string? Compose(object? value)
    {
        List<string> lines = new List<string> { "" };
        bool skipping = false;
        void Visit(object? item, string? indent = null)
        {
            // Traverse fragments directly: ToString would lose missing-line and indentation behavior.
            if (item is TemplateContent templateContent) { foreach (object? fragment in templateContent.Fragments) Visit(fragment, indent); return; }
            if (item is object?[] sequence) { foreach (object? child in sequence) Visit(child, indent); return; }
            if (item is Data data)
            {
                if (data.Items is not null)
                {
                    List<string> scalars = new();
                    bool scalarList = true;
                    foreach (Data child in data.Items)
                    {
                        if (!child.Values.TryGetValue("_", out object? scalar) || !Condition(scalar)) { scalarList = false; break; }
                        scalars.Add(Scalar(scalar!));
                    }
                    // Raw object-data lists retain their layout. A dictionary contributes
                    // no text unless its scalar '_' value can be interpolated.
                    item = scalarList ? string.Join(", ", scalars) : new RenderedList(new List<object?>(data.Items));
                }
                else if (data.Values.TryGetValue("_", out object? scalar) && Condition(scalar)) item = scalar;
                else return;
            }
            if (item is Arguments arguments) item = new RenderedList(arguments.Items);
            if (item is RenderedList rendered)
            {
                if (rendered.Items.Count == 0) return;
                bool emptyLine = lines[^1] == "";
                string lastIndent = Regex.Match(lines[^1], @"^[ \t]*").Value;
                string? nextIndent = "   " + (indent ?? lastIndent);
                string preview = Compose(rendered.Items[0]) ?? "";
                if (Regex.Match(preview, @"^[ \t]*").Length > lastIndent.Length) nextIndent = null;
                bool incomplete = lines[^1] == indent;
                if (incomplete) nextIndent = indent;
                else if (lastIndent == lines[^1]) { nextIndent = lastIndent; incomplete = true; }
                bool first = true;
                foreach (object? child in rendered.Items)
                {
                    object? content = child;
                    if (preview.StartsWith('\n')) nextIndent = indent;
                    else if (!(incomplete && first) && (!emptyLine || !first))
                    {
                        Visit("\n" + nextIndent);
                        if (!string.IsNullOrEmpty(nextIndent) && content is string text)
                            content = text.Replace("\n", "\n" + nextIndent);
                    }
                    Visit(content, nextIndent);
                    first = false;
                }
                return;
            }
            if (item is Missing missing) item = missing.Replacement;
            if (item is null) { lines[^1] = ""; skipping = true; return; }
            string[] parts = Scalar(item).Split('\n');
            for (int i = 0; i < parts.Length; i++)
            {
                if (i == 0 && skipping)
                {
                    if (parts.Length == 1) return;
                    skipping = false;
                }
                else
                {
                    lines[^1] += parts[i];
                    if (i < parts.Length - 1) lines.Add("");
                }
            }
        }
        Visit(value);
        if (skipping)
        {
            if (lines.Count == 1) return null;
            lines.RemoveAt(lines.Count - 1);
        }
        return string.Join("\n", lines);
    }
}

public static class GeneratedTemplateProgram
{
    public static int Main(string[] args)
    {
        try
        {
            string? template = null, dataFile = null, dataRoot = null;
            for (int i = 0; i < args.Length; i++)
            {
                if (args[i] == "--template" && template is null && i + 1 < args.Length) template = args[++i];
                else if (args[i] == "--data-file" && dataFile is null && i + 1 < args.Length) dataFile = args[++i];
                else if (args[i] == "--data-root" && dataRoot is null && i + 1 < args.Length) dataRoot = args[++i];
                else if (args[i] == "--help") { Console.WriteLine("Usage: --template NAME [--data-file FILE] [--data-root DIRECTORY]"); return 0; }
                else throw new ArgumentException("Unknown or incomplete argument: " + args[i]);
            }
            if (template is null) throw new ArgumentException("Specify --template NAME.");
            using JsonDocument? data = dataFile is null ? null : JsonDocument.Parse(File.ReadAllText(dataFile));
            string result = GeneratedTemplates.Render(template, data?.RootElement, dataRoot);
            Console.Write(result);
            return 0;
        }
        catch (Exception error) { Console.Error.WriteLine(error.Message); return 1; }
    }
}
