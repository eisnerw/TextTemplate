internal sealed class BulletIndent
{
    internal int Level, Index;
    internal string Indent;
    internal BulletIndent? Parent;
    internal int? InitializerLevel;
    internal BulletIndent(string indent, BulletIndent? current, int level)
    {
        Indent = indent;
        if (current is null) { Level = level; return; }
        if (indent == current.Indent)
        {
            Level = current.Level; Index = current.Index + 1; Parent = current.Parent;
            InitializerLevel = current.InitializerLevel; return;
        }
        for (BulletIndent? match = current.Parent; match is not null; match = match.Parent)
            if (match.Level == level || indent == match.Indent)
            {
                Level = match.Level; Index = match.Index + 1; Parent = match.Parent;
                InitializerLevel = Level; return;
            }
        Level = level; Parent = current; InitializerLevel = current.InitializerLevel;
    }
    internal string Format(IReadOnlyList<string>? styles)
    {
        if (styles is null || styles.Count == 0) return "(" + Level + "." + Index + ")";
        const string token = @"i\:[ivxldcm]+|i|I\:[IVXLDCM]+|I|1\:[0-9]+|1|a\:[a-z]+|a|A\:[A-Z]+|A";
        string styleText = styles[Math.Min(Level, styles.Count - 1)];
        string[] parts = Regex.Replace(styleText, "(.*?(" + token + @").\S*?)", "$1\x02").Split('\x02');
        string combined = "";
        BulletIndent? current = this;
        for (int i = parts.Length - 1; i >= 0 && current is not null; i--)
        {
            string style = parts[i];
            string padding = Regex.Match(style, "^ +").Value;
            style = style[padding.Length..];
            string prefix = "", postfix = "", kind = "";
            Match match = Regex.Match(style, "^(.*?)(" + token + ")(.*)$");
            if (match.Success)
            {
                prefix = match.Groups[1].Value; style = match.Groups[2].Value; postfix = match.Groups[3].Value;
                kind = style[..1];
                if (style.Contains(':'))
                    style = InitializerLevel is not null && current.Level > InitializerLevel ? kind : style[(style.IndexOf(':') + 1)..];
            }
            else if (style.Length > 1)
            {
                if ("(<#$%*.-=+`~[{_=+|'\"".Contains(style[0])) { prefix = style[..1]; style = style[1..]; }
                if (style.Length > 0 && ")>*]}.`~*-_=+|:'\"".Contains(style[^1])) { postfix = style[^1..]; style = style[..^1]; }
            }
            string bullet = style;
            if (kind.Length == 1)
            {
                int start;
                switch (kind)
                {
                    case "I": case "i":
                        start = style == kind ? 1 : FromRoman(style);
                        bullet = ToRoman(current.Index + start);
                        if (kind == "i") bullet = bullet.ToLowerInvariant();
                        break;
                    case "1":
                        bullet = (current.Index + (style == "1" ? 1 : EvaluationContext.ParseInteger(style))).ToString(CultureInfo.InvariantCulture);
                        break;
                    case "a": case "A":
                        start = style.ToLowerInvariant() == "a" ? 1 : FromAlphabet(style);
                        bullet = ToAlphabet(current.Index + start);
                        if (kind == "a") bullet = bullet.ToLowerInvariant();
                        break;
                }
                if (padding.Length > 0 && bullet.Length < padding.Length + 1)
                    prefix = padding[..(padding.Length - bullet.Length + 1)] + prefix;
                bullet = prefix + bullet + postfix;
                current = current.Parent;
            }
            combined = bullet + combined;
        }
        return combined;
    }
    private static int FromRoman(string value)
    {
        int Digit(char c) => c switch { 'I' => 1, 'V' => 5, 'X' => 10, 'L' => 50, 'C' => 100, 'D' => 500, 'M' => 1000, _ => -1 };
        value = value.ToUpperInvariant();
        int result = 0;
        for (int i = 0; i < value.Length; i++)
        {
            int digit = Digit(value[i]);
            if (digit < 0) return -1;
            int next = i + 1 < value.Length ? Digit(value[i + 1]) : -1;
            if (digit < next) { result += next - digit; i++; } else result += digit;
        }
        return result;
    }
    private static string ToRoman(int value)
    {
        // The legacy routine uses bitwise integer division and alternating bases 5/2.
        string text = "";
        for (int a = 5, b = 0; value != 0; b++, a ^= 7)
        {
            int remainder = value % a;
            value /= a;
            while (remainder-- > 0)
            {
                int index = b;
                if (remainder > 2)
                {
                    int old = value; value &= -2; remainder = 1;
                    index = b + old - value + remainder;
                }
                text = (index < 7 ? "IVXLCDM"[index].ToString() : "undefined") + text;
            }
        }
        return text;
    }
    private static int FromAlphabet(string value)
    {
        int result = 0;
        foreach (char c in value.ToUpperInvariant()) result = result * 26 + c - 64;
        return result;
    }
    private static string ToAlphabet(int value)
    {
        string result = "";
        while (value > 0) { value--; result = (char)('A' + value % 26) + result; value /= 26; }
        return result;
    }
}
